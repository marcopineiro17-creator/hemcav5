<script>
(function () {
  "use strict";

  var raiz = document.getElementById("cpm-divisiones");
  if (!raiz || raiz.getAttribute("data-init")) return;
  raiz.setAttribute("data-init", "arrancando");

  /* Si algo falla, se retira el ocultado y todo el contenido queda visible. */
  function rendirse(e) {
    try {
      raiz.classList.remove("js-on");
      var i = raiz.querySelector(".intro");
      if (i) i.classList.add("is-gone");
    } catch (_) {}
    try { if (window.console && console.warn) console.warn("[CPM] modo seguro:", e); } catch (_) {}
  }
  function safe(fn) { try { fn(); } catch (e) {} }

  try { principal(); } catch (e) { rendirse(e); }

  function principal() {

/*__MOTOR__*/

    var M = hbEngine(raiz, {
      claseVisible: "is-visible",
      limite: 9000,
      /* El runtime global de CPM ya reconoce #cpm-divisiones al escanear, y
         ademas atiende este tipo de mensaje. No se manda el saludo de
         Hummingbird: ese es el que crea el boton flotante de WhatsApp, y este
         bloque no lo lleva. */
      tipoAltura: "cpm-divisiones:height",
      proyecto: "divisiones"
    });

    /* =============================================================
       1. ENLACES QUE SALEN DEL BLOQUE
       -------------------------------------------------------------
       El problema original: los enlaces eran relativos (/contacto) y el
       bloque vive dentro de un iframe. Una ruta relativa se resuelve contra
       la direccion del documento del iframe, no contra la del sitio, y la
       navegacion ocurria DENTRO del marco. De ahi el "rechazo la conexion".

       Se arregla en dos pasos:
         a) La direccion se vuelve absoluta.
         b) El enlace se abre en la ventana completa (target _top), no en el
            marco. Con un respaldo por JS por si el target no basta.

       Y para no depender de que las rutas escritas a mano sean las buenas,
       primero se busca cada destino EN EL MENU REAL del sitio: el documento
       padre es legible, asi que se lee su navegacion y se enlaza justo donde
       enlaza el propio sitio. Si no aparece, se usa la ruta de respaldo.
       ============================================================= */
    var ORIGEN = "https://www.cpmempresarial.com";
    safe(function () {
      /* Mejor el origen real: asi funciona igual en un dominio de pruebas. */
      var o = window.parent.location.origin;
      if (o && /^https?:/.test(o)) ORIGEN = o;
    });

    function sinAcentos(t) {
      t = (t || "").toLowerCase();
      try { t = t.normalize("NFD").replace(/[\u0300-\u036f]/g, ""); } catch (e) {}
      return t.replace(/\s+/g, " ").trim();
    }

    /* Rutas del menu del sitio, indexadas por la palabra que las nombra. */
    var menu = null;
    function leeMenu() {
      if (menu) return menu;
      menu = {};
      safe(function () {
        var docPadre = window.parent.document;
        var anclas = docPadre.querySelectorAll('a[href]');
        for (var i = 0; i < anclas.length; i++) {
          var a = anclas[i];
          var href = a.getAttribute("href") || "";
          if (/^(#|mailto:|tel:|javascript:)/i.test(href)) continue;
          var u;
          try { u = new URL(href, docPadre.baseURI); } catch (e) { continue; }
          if (u.origin !== ORIGEN) continue;                 /* nada externo */
          var ruta = u.pathname.replace(/\/+$/, "");
          if (!ruta || ruta === "/divisiones") continue;      /* la pagina actual */
          var texto = sinAcentos(a.textContent);
          if (!texto || texto.length > 40) continue;
          /* Paginas de tramite: no son destinos de division y sus nombres se
             parecen demasiado ("aviso legal" no es la division legal). */
          if (/aviso|privacidad|terminos|cookies|politica|blog/.test(texto)) continue;
          /* Se queda con la primera aparicion: el menu principal va antes
             que el pie, y suele ser el enlace canonico. */
          if (!menu[texto]) menu[texto] = u.href;
        }
      });
      return menu;
    }

    /* Busca el destino en el menu. Coincidencia exacta o por el principio del
       texto; nada de buscar la palabra en cualquier posicion, que es como se
       acaba enlazando la division legal al aviso legal. */
    function rutaDelSitio(claves) {
      var m = leeMenu();
      for (var i = 0; i < claves.length; i++) {
        var k = sinAcentos(claves[i]);
        if (m[k]) return m[k];
      }
      for (var j = 0; j < claves.length; j++) {
        var k2 = sinAcentos(claves[j]);
        for (var texto in m) {
          if (Object.prototype.hasOwnProperty.call(m, texto) && texto.indexOf(k2) === 0) return m[texto];
        }
      }
      return null;
    }

    var externos = [].slice.call(raiz.querySelectorAll("a[data-cpm-destino]"));
    externos.forEach(function (a) {
      var claves = (a.getAttribute("data-cpm-destino") || "").split("|");
      var respaldo = a.getAttribute("data-cpm-respaldo") || "/";
      var url = rutaDelSitio(claves) || (ORIGEN + respaldo);
      a.setAttribute("href", url);
      a.setAttribute("target", "_top");
      a.setAttribute("rel", "noopener");
    });

    /* Respaldo: si el navegador ignorase el target (pasa en algunos marcos
       con sandbox), se navega la ventana completa a mano. */
    safe(function () {
      raiz.addEventListener("click", function (ev) {
        var a = ev.target.closest ? ev.target.closest('a[target="_top"]') : null;
        if (!a || ev.defaultPrevented || ev.metaKey || ev.ctrlKey || ev.button > 0) return;
        var url = a.getAttribute("href");
        if (!url || url.charAt(0) === "#") return;
        ev.preventDefault();
        try { window.top.location.href = url; }
        catch (e) { window.open(url, "_blank", "noopener"); }
      });
    });

    /* ---------- 2. Portada de entrada ----------
       Se cierra sola por CSS. Aqui solo el boton de omitir y el corte cuando
       el bloque ni siquiera esta a la vista al cargar. */
    var portada = raiz.querySelector("#cpmIntro");
    safe(function () {
      var boton = raiz.querySelector(".intro-skip");
      if (boton && portada) {
        boton.addEventListener("click", function () {
          portada.classList.add("is-gone");
          M.programarAjuste();
        });
      }
      if (portada) {
        /* Si al cargar ya estas mas abajo, la entrada no tiene sentido. */
        if (M.techo(portada, M.desfase()) < -80) portada.classList.add("is-gone");
        portada.addEventListener("animationend", function (ev) {
          if (ev.animationName === "cpmIntroFuera") {
            portada.classList.add("is-gone");
            M.programarAjuste();
          }
        });
      }
    });

    /* ---------- 3. Capitulo activo y menu ---------- */
    var enlaces   = [].slice.call(raiz.querySelectorAll("[data-chapter]"));
    var capitulos = [].slice.call(raiz.querySelectorAll(".chapter[data-theme]"));
    var actual = "";
    function activar(id) {
      if (!id || id === actual) return;
      actual = id;
      enlaces.forEach(function (l) { l.classList.toggle("is-active", l.getAttribute("data-chapter") === id); });
      capitulos.forEach(function (s) { s.classList.toggle("is-active", s.id === id); });
    }
    if (capitulos.length) {
      M.continuo(function (H, S) {
        var mejor = null, menor = Infinity;
        for (var i = 0; i < capitulos.length; i++) {
          var s = capitulos[i];
          var arriba = M.techo(s, S);
          var abajo  = arriba + s.getBoundingClientRect().height;
          var d = Math.abs(arriba - H * 0.38);
          if (arriba < H * 0.72 && abajo > H * 0.24 && d < menor) { menor = d; mejor = s; }
        }
        if (mejor) activar(mejor.id);
      });
    }

    /* ---------- 4. Entradas ---------- */
    [].slice.call(raiz.querySelectorAll(".reveal")).forEach(function (el) { M.alEntrar(el); });

    /* ---------- 5. Enlaces internos del propio bloque ----------
       scrollIntoView si atraviesa el marco: arrastra el scroll del padre. */
    safe(function () {
      raiz.addEventListener("click", function (ev) {
        var a = ev.target.closest ? ev.target.closest('a[href^="#"]') : null;
        if (!a) return;
        var destino = raiz.querySelector(a.getAttribute("href"));
        if (!destino) return;
        ev.preventDefault();
        if (portada) portada.classList.add("is-gone");
        try { destino.scrollIntoView({ behavior: M.lento ? "auto" : "smooth", block: "start" }); }
        catch (e) { try { destino.scrollIntoView(); } catch (e2) {} }
      });
    });

    /* ---------- 6. Ano del pie ---------- */
    safe(function () {
      var y = raiz.querySelector("#cpmYear");
      if (y) y.textContent = String(new Date().getFullYear());
    });

    raiz.setAttribute("data-init", "listo");
    M.arrancar();
  }
})();
</script>
