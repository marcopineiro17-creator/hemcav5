<script>
(function () {
  'use strict';
  var root = document.getElementById('hbp');
  if (!root || root.getAttribute('data-init')) return;
  root.setAttribute('data-init', 'starting');

  var q = function (s, c) { return Array.prototype.slice.call((c || root).querySelectorAll(s)); };
  var one = function (s, c) { return (c || root).querySelector(s); };
  var numAttr = function (el, a, d) { var v = parseFloat(el.getAttribute(a)); return isNaN(v) ? d : v; };

  /* Si algo falla, se retira el ocultado y todo el contenido queda visible. */
  function rendirse(err) {
    try {
      root.classList.remove('motion-ready');
      q('.reveal').forEach(function (el) { el.classList.add('is-visible'); el.style.opacity = ''; el.style.transform = ''; });
      root.setAttribute('data-init', 'fallback');
    } catch (e) {}
    try { if (window.console && console.warn) console.warn('[HBP] modo seguro:', err); } catch (e) {}
  }
  function safe(fn) { try { fn(); } catch (e) {} }

  try { principal(); } catch (err) { rendirse(err); }

  function principal() {

/*__MOTOR__*/

    var M = hbEngine(root, {
      claseVisible: 'is-visible',
      limite: 9000,
      /* Sin medida fiable (iframe de otro origen): nada de scroll, todo
         visible, y el carrusel del teléfono pasa a tiempo y clic. */
      alRendirse: function () { root.classList.add('sin-scroll'); porTiempo(); }
    });

    var menuButton = one('#hbpMenuButton'), menu = one('#hbpMenu');
    var nav = one('#hbpNav'), progress = one('#hbpProgress');
    var steps = q('.story-step'), slides = q('.story-slide'), bars = q('.story-bars i');
    var brand = one('#hbpPhoneBrand'), caption = one('#hbpPhoneCaption');
    var activeSlide = -1;
    var names = ['Marisquería Poseidón', 'Distribuidora Serrano', 'Pulatu Café', 'Amorcito Corazón'];

    /* ---------- 1. Interacción básica: funciona siempre ---------- */
    function setMenu(open) {
      if (!menu) return;
      menu.classList.toggle('open', open);
      menu.setAttribute('aria-hidden', open ? 'false' : 'true');
      if (menuButton) {
        menuButton.setAttribute('aria-expanded', open ? 'true' : 'false');
        menuButton.setAttribute('aria-label', open ? 'Cerrar menú' : 'Abrir menú');
      }
    }
    if (menuButton) menuButton.addEventListener('click', function () { setMenu(!menu.classList.contains('open')); });
    q('#hbpMenu a').forEach(function (a) { a.addEventListener('click', function () { setMenu(false); }); });
    document.addEventListener('keydown', function (e) { if (e.key === 'Escape') setMenu(false); });

    /* scrollIntoView arrastra también el scroll del documento padre. */
    q('a[href^="#hbp"]').forEach(function (a) {
      a.addEventListener('click', function (e) {
        var t; try { t = document.querySelector(a.getAttribute('href')); } catch (err) { return; }
        if (!t) return;
        e.preventDefault();
        try { t.scrollIntoView({ behavior: M.lento ? 'auto' : 'smooth', block: 'start' }); }
        catch (err) { try { t.scrollIntoView(); } catch (e2) {} }
      });
    });

    var year = one('#hbpYear'); if (year) year.textContent = (new Date()).getFullYear();
    q('svg').forEach(function (s) { s.setAttribute('focusable', 'false'); });

    /* ---------- 2. Carrusel de casos ---------- */
    function selectSlide(index) {
      if (!slides.length) return;
      index = Math.max(0, Math.min(slides.length - 1, index));
      if (index === activeSlide) return;
      activeSlide = index;
      slides.forEach(function (s, i) { s.classList.toggle('active', i === index); });
      steps.forEach(function (s, i) {
        s.classList.toggle('active', i === index);
        s.setAttribute('aria-current', i === index ? 'step' : 'false');
      });
      bars.forEach(function (b, i) { b.classList.toggle('on', i === index); b.classList.toggle('done', i < index); });
      if (brand) brand.textContent = slides[index].getAttribute('data-brand') || names[index] || '';
      if (caption) caption.textContent = names[index] || '';
    }
    steps.forEach(function (step, i) {
      step.addEventListener('click', function () { selectSlide(i); });
      step.addEventListener('focus', function () { selectSlide(i); });
      step.addEventListener('keydown', function (e) {
        if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); selectSlide(i); }
      });
    });
    selectSlide(0);

    /* Plan B del carrusel cuando no hay scroll medible. */
    function porTiempo() {
      if (slides.length < 2 || M.lento) return;
      setInterval(function () { selectSlide((activeSlide + 1) % slides.length); }, 4200);
    }

    /* ---------- 3. Estado del nav y barra de progreso ---------- */
    var navLinks = q('.nav-links a');
    var anclas = ['hbp-work', 'hbp-social', 'hbp-method'].map(function (id) { return document.getElementById(id); });
    M.continuo(function (H, S) {
      var r = root.getBoundingClientRect();
      var top = r.top + S, total = Math.max(1, r.height - H);
      var p = M.limita(-top / total);
      if (progress) progress.style.transform = 'scaleX(' + p.toFixed(4) + ')';
      if (nav) nav.classList.toggle('scrolled', top < -35);
      var act = '';
      anclas.forEach(function (s) { if (s && M.techo(s, S) < H * 0.36) act = s.id; });
      navLinks.forEach(function (a) { a.classList.toggle('active', a.getAttribute('href') === '#' + act); });
    });

    /* ---------- 4. Modo sin movimiento ---------- */
    if (M.lento) {
      root.setAttribute('data-init', 'sin-movimiento');
      M.arrancar();
      return;
    }
    root.classList.add('motion-ready');

    /* ---------- 5. Entradas ---------- */
    q('.reveal').forEach(function (el) { M.alEntrar(el); });

    /* ---------- 6. Marcos que se abren con clip-path ---------- */
    q('.media').forEach(function (el) {
      M.continuo(function (H, S) {
        var r;
        try { r = el.getBoundingClientRect(); } catch (e) { return; }
        var p = M.limita((H - (r.top + S)) / (H + r.height));
        var e = 1 - M.limita((H * 0.95 - (r.top + S)) / (H * 0.57));
        var y = (7 * e).toFixed(2), x = (8 * e).toFixed(2), rad = (30 * e).toFixed(1);
        var cp = 'inset(' + y + '% ' + x + '% ' + y + '% ' + x + '% round ' + rad + 'px)';
        el.style.clipPath = cp; el.style.webkitClipPath = cp;
        /* El parallax va por variable CSS: la regla de .media img ya la usa. */
        var amount = numAttr(el, 'data-parallax', 0);
        if (amount) el.style.setProperty('--parallax-y', (-7 + amount * (p - 0.5)).toFixed(2) + '%');
      });
    });

    /* ---------- 7. Parallax del hero y la cinta ---------- */
    (function () {
      var stage = one('.hero-stage');
      var frames = q('.hero-frame');
      if (stage && frames.length) {
        M.continuo(function (H, S) {
          var r = stage.getBoundingClientRect();
          var p = M.limita((H - (r.top + S)) / (H + r.height));
          frames.forEach(function (el, i) {
            var y = ((i + 1) * -9) * (p - 0.5) * 2;
            var rot = (i === 1 ? -9 : i === 2 ? 10 : 0) * (p - 0.5) * 2;
            el.style.transform = 'translateY(' + y.toFixed(2) + '%) rotate(' + rot.toFixed(2) + 'deg)';
          });
        });
      }
      var ticker = one('#hbpTicker'), tBox = one('.client-ticker');
      if (ticker && tBox) {
        M.continuo(function (H, S) {
          var r = tBox.getBoundingClientRect();
          var p = M.limita((H - (r.top + S)) / (H + r.height));
          ticker.style.transform = 'translateX(' + (-p * 22).toFixed(2) + '%)';
        });
      }
      var aura = one('.phone-aura'), social = document.getElementById('hbp-social');
      if (aura && social) {
        M.continuo(function (H, S) {
          var r = social.getBoundingClientRect();
          var p = M.limita((H - (r.top + S)) / (H + r.height));
          aura.style.transform = 'rotate(' + (p * 190).toFixed(1) + 'deg)';
        });
      }
    })();

    /* ---------- 8. Teléfono anclado + caso activo según el scroll ----------
       Antes era position:sticky. Dentro del iframe el documento no hace
       scroll propio, así que el sticky no puede funcionar: el anclaje se
       calcula con translateY sobre coordenadas del viewport real. */
    (function () {
      var layout = one('#hbpStoryLayout'), stage = one('#hbpPhoneStage');
      var rail = one('#hbpPhoneRail') || layout;
      if (!layout || !stage) return;
      root.classList.add('telefono-js');
      var ultimaY = null;
      M.continuo(function (H, S) {
        /* En móvil el teléfono va en el flujo (ver el CSS): nada que anclar.
           Anclarlo obligaba a superponer el texto, y el fondo sólido que el
           texto necesita para ser legible acababa tapándolo por completo. */
        if (window.innerWidth <= 760) { stage.style.transform = ''; return; }
        /* Se mide sobre el CARRIL, que es donde vive el teléfono, no sobre
           la rejilla completa: con la rejilla el recorrido no coincidía y el
           teléfono se iba quedando abajo. */
        var arriba = rail.getBoundingClientRect().top + S;
        var altoEsc = stage.offsetHeight;
        var rec = Math.max(0, rail.offsetHeight - altoEsc);
        /* Centrado vertical en pantalla mientras está anclado. */
        /* En movil el telefono va en la banda superior; en escritorio,
           centrado en pantalla. */
        var hueco = (window.innerWidth <= 760) ? 12 : Math.max(16, (H - altoEsc) / 2);
        var y = Math.min(Math.max(hueco - arriba, 0), rec);
        /* Redondeo a medio píxel: los valores fraccionarios hacían temblar
           el render en algunos navegadores. */
        y = Math.round(y * 2) / 2;
        if (y !== ultimaY) {
          ultimaY = y;
          stage.style.transform = 'translate3d(0,' + y + 'px,0)';
        }
        /* El caso activo es el paso más cercano al centro de la pantalla.
           Con zona muerta: sólo cambia si el nuevo candidato está
           claramente más cerca. Sin esto, dos pasos a distancia parecida
           se alternaban en cada fotograma y la imagen parpadeaba. */
        if (steps.length) {
          var mejor = 0, dist = Infinity, distActual = Infinity;
          for (var i = 0; i < steps.length; i++) {
            var sr = steps[i].getBoundingClientRect();
            var d = Math.abs((sr.top + sr.bottom) / 2 + S - H * 0.5);
            if (i === activeSlide) distActual = d;
            if (d < dist) { dist = d; mejor = i; }
          }
          if (mejor !== activeSlide && dist < distActual - H * 0.08) selectSlide(mejor);
        }
      });
    })();

    /* ---------- 9. Detalles de puntero ---------- */
    q('.liquid').forEach(function (el) {
      el.addEventListener('pointermove', function (e) {
        var r = el.getBoundingClientRect();
        el.style.setProperty('--mx', ((e.clientX - r.left) / r.width * 100) + '%');
        el.style.setProperty('--my', ((e.clientY - r.top) / r.height * 100) + '%');
      });
    });
    q('.magnetic').forEach(function (el) {
      el.style.transition = 'transform .35s cubic-bezier(.16,1,.3,1)';
      el.addEventListener('pointermove', function (e) {
        if (window.innerWidth < 900) return;
        var r = el.getBoundingClientRect();
        el.style.transform = 'translate(' + ((e.clientX - r.left - r.width / 2) * 0.12).toFixed(1) + 'px,' +
                                            ((e.clientY - r.top - r.height / 2) * 0.18).toFixed(1) + 'px)';
      });
      el.addEventListener('pointerleave', function () { el.style.transform = ''; });
    });

    M.arrancar();
    root.setAttribute('data-init', M.enMarco ? (M.legible ? 'iframe-medible' : 'iframe-ciego') : 'directo');
  }
})();
</script>
